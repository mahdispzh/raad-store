"use client";

import {useState} from "react";
import {useForm} from "react-hook-form";
import {zodResolver} from "@hookform/resolvers/zod";
import {z} from "zod";
import {CustomInputRHF} from "@/src/app/dashboard/products/_components/custom-input";
import {SelectRHF} from "@/src/app/dashboard/products/_components/custom-select";

// Schema نمونه (می‌تونی بعداً جایگزین خودت کنی)
const formSchema = z.object({
  title: z.string().min(3, "حداقل 3 کاراکتر وارد کنید"),
  category: z.string().min(1, "انتخاب دسته الزامی است"),
  price: z.coerce.number().min(1, "قیمت باید بیشتر از صفر باشد"),
  stock: z.coerce.number().min(0, "موجودی نمی‌تواند منفی باشد"),
  description: z.string().min(5, "حداقل 5 کاراکتر وارد کنید"),

});

type FormValues = z.infer<typeof formSchema>;

// تعریف فیلدهای هر مرحله
const STEP_FIELDS = [
  ["title", "category"],
  ["price", "stock"],
  ["description"],
];

const options = [
  {value: 1, label: "موبایل"},
  {value: 2, label: "لپ تاپ"},
  {value: 3, label: "هدفون"},
];

export default function MultiStepForm() {
  const [step, setStep] = useState(0);

  const form = useForm<FormValues>({
    defaultValues: { categoryId: undefined },
    resolver: zodResolver(formSchema),
    mode: "onChange",
    defaultValues: {
      title: "",
      category: "",
      price: "",
      stock: "",
      description: "",
    },
  });

  const {handleSubmit, trigger, control} = form;

  const nextStep = async () => {
    const currentFields = STEP_FIELDS[step];
    const valid = await trigger(currentFields);
    if (!valid) return;
    setStep((prev) => prev + 1);
  };

  const previousStep = () => {
    setStep((prev) => prev - 1);
  };

  const onSubmit = (data: FormValues) => {
    console.log("Submit نهایی:", data);
    alert("فرم با موفقیت ثبت شد!");
  };


  return (
    <div className="max-w-md mx-auto p-6 border rounded shadow-md flex flex-col gap-4">
      <h2 className="text-lg font-bold mb-4">فرم سه مرحله‌ای</h2>

      {/* مرحله 1 */}
      {step === 0 && (
        <>
          <CustomInputRHF
            control={control}
            name="title"
            label="عنوان محصول"
            placeholder="عنوان را وارد کنید"
          />

          <CustomInputRHF
            control={control}
            name="category"
            label="دسته‌بندی"
            placeholder="دسته‌بندی را وارد کنید"
          />




        </>
  )
}

{/* مرحله 2 */
}
{
  step === 1 && (
    <>
      <CustomInputRHF
        control={control}
        name="price"
        type="number"
        label="قیمت"
        placeholder="مثلاً 50000"
      />
      <CustomInputRHF
        control={control}
        name="stock"
        type="number"
        label="موجودی"
        placeholder="مثلاً 10"
      />
    </>
  )
}

{/* مرحله 3 */
}
{
  step === 2 && (
    <>
      <CustomInputRHF
        control={control}
        name="description"
        label="توضیحات"
        placeholder="توضیحات محصول را وارد کنید"
      />
    </>
  )
}

{/* دکمه‌ها */
}
  <div className="flex justify-between mt-4">
    {step > 0 && (
      <button
        type="button"
        onClick={previousStep}
        className="px-4 py-2 bg-gray-200 rounded"
      >
        قبلی
      </button>
    )}

    {step < STEP_FIELDS.length - 1 ? (
      <button
        type="button"
        onClick={nextStep}
        className="px-4 py-2 bg-blue-500 text-white rounded"
      >
        مرحله بعد
      </button>
    ) : (
      <button
        type="button"
        onClick={handleSubmit(onSubmit)}
        className="px-4 py-2 bg-green-500 text-white rounded"
      >
        ثبت نهایی
      </button>
    )}
  </div>
</div>
)
  ;
}
